def _mask_key(k: Optional[str]) -> Optional[str]:
    if not k:
        return None
    s = str(k).strip()
    return s if len(s) <= 12 else (s[:6] + "..." + s[-4:])

def _check_env_file_permissions(env_path: Path) -> bool:
    try:
        st = env_path.stat()
        # refuse if world-writable (POSIX check); Windows will not provide same bits but it's fine
        if bool(st.st_mode & 0o022):
            logger.error(".env has insecure permissions (world-writable). Refusing to apply.")
            return False
        return True
    except Exception as e:
        logger.warning(f"Could not stat .env: {e}")
        return False

def _verify_env_hmac(env_path: Path, secret: str) -> bool:
    try:
        hmac_path = env_path.with_name(env_path.name + ".hmac")
        if not hmac_path.exists():
            logger.error(".env.hmac missing")
            return False
        expected = hmac_path.read_text().strip()
        data = env_path.read_bytes()
        calc = hmac.new(secret.encode('utf-8'), data, hashlib.sha256).hexdigest()
        if not hmac.compare_digest(calc, expected):
            logger.error(".env HMAC mismatch")
            return False
        return True
    except Exception as e:
        logger.exception(f".env HMAC verify error: {e}")
        return False

# VoiceQueueItem structure
class _VoiceRequest:
    def __init__(self, text: str, voice_id: Optional[str] = None, priority: int = 5, truncate_length: int = 600):
        self.text = text
        self.voice_id = voice_id
        self.priority = priority

        self.timestamp = time.time()
        self.attempts = 0

# Actual manager
class ElevenHybridVoiceManager:
    """
    Manages a queue of TTS requests, generates audio via ElevenLabs SDK,
    and plays audio using simpleaudio (if available) for interruptible playback.
    """
    def __init__(self, api_key: str):
        self._api_key = api_key
        self._queue = queue.PriorityQueue()
        self._worker = threading.Thread(target=self._worker_loop, daemon=True, name="ElevenVoiceWorker")
        self._stop_event = threading.Event()
        self._current_playback = None
        self._playback_lock = threading.Lock()
        self._simpleaudio = None
        self._eleven_generate = None
        self._eleven_play = None
        self._eleven_set_key = None
        self._setup_success = False
        self._init_sdk()
        if self._setup_success:
            self._worker.start()

    def _init_sdk(self):
        try:
            from elevenlabs import set_api_key, generate, play  # type: ignore
            try:
                set_api_key(self._api_key)
            except Exception as e:
                logger.warning(f"ElevenLabs:set_api_key failed: {e}")
            self._eleven_generate = generate
            self._eleven_play = play
            self._eleven_set_key = set_api_key
            try:
                import simpleaudio as sa  # type: ignore
                self._simpleaudio = sa
                logger.debug("simpleaudio available — will use interruptible PCM playback.")
            except Exception:
                self._simpleaudio = None
                logger.debug("simpleaudio unavailable — using official SDK.play fallback (may be non-interruptible).")
            self._setup_success = True
            logger.info("ElevenHybridVoiceManager initialized.")
        except Exception as e:
            logger.exception(f"ElevenLabs SDK init error: {e}")
            self._setup_success = False

    def stop_playback(self):
        with self._playback_lock:
            self._stop_event.set()
            play_obj = self._current_playback
            if play_obj and self._simpleaudio:
                try:
                    play_obj.stop()
                except Exception:
                    logger.debug("Failed to stop simpleaudio playback (ignored).")

    def cleanup(self):
        self.stop_playback()
        self._stop_event.set()
        # wake worker by sending sentinel
        try:
            self._queue.put((9999, time.time(), None))
        except Exception:
            pass
        logger.debug("ElevenHybridVoiceManager cleanup requested.")

    def enqueue(self, req: _VoiceRequest):
        # priority queue by (priority, timestamp)
        self._queue.put((req.priority, req.timestamp, req))
        logger.debug("Enqueued TTS request (priority=%s, len=%s)", req.priority, len(req.text))

    def _truncate_text(self, text: str, length: int) -> str:
        if not text:
            return ""
        if len(text) <= length:
            return text
        # try to cut at sentence boundary
        cut = text[:length].rsplit('.', 1)[0]
        if not cut.strip():
            cut = text[:length]
        return cut.strip() + " ... (truncated for speech)"

    def _worker_loop(self):
        logger.debug("Eleven voice worker started.")
        backoff_base = 1.2
        while not self._stop_event.is_set():
            try:
                prio, ts, item = self._queue.get(timeout=0.5)
            except queue.Empty:
                continue
            if item is None:
                break
            if not isinstance(item, _VoiceRequest):
                continue
            item.attempts += 1
            try:
                text = item.text.strip()
                if not text:
                    continue
                text = self._truncate_text(text, item.truncate_length)
                try:
                    if self._eleven_generate:
                        # --- PATCH: lock voice to ELEVEN_VOICE_ID (no alloy fallback)
                        audio_obj = self._eleven_generate(
                            text=text,
                            voice=(item.voice_id or ELEVEN_VOICE_ID),
                            model="eleven_monolingual_v1"
                        )
                    else:
                        raise RuntimeError("ElevenLabs generate unavailable")
                except Exception as e:
                    logger.warning("TTS generate failed (attempt %s): %s", item.attempts, e)
                    if item.attempts < 4:
                        delay = (backoff_base ** item.attempts) + random.uniform(0, 0.5)
                        time.sleep(delay)
                        self._queue.put((item.priority + 1, time.time(), item))
                        continue
                    else:
                        logger.error("TTS generate permanently failed after attempts: %s", e)
                        continue

                if self._stop_event.is_set():
                    logger.debug("TTS worker interrupted before playback start.")
                    continue

                played = False
                # Attempt to extract raw bytes
                raw_bytes = None
                if isinstance(audio_obj, (bytes, bytearray)):
                    raw_bytes = bytes(audio_obj)
                else:
                    for attr in ("content", "audio", "raw", "read"):
                        if hasattr(audio_obj, attr):
                            try:
                                candidate = getattr(audio_obj, attr)
                                raw_bytes = candidate() if callable(candidate) else candidate
                                break
                            except Exception:
                                continue
                    if raw_bytes is None and hasattr(audio_obj, "read"):
                        try:
                            raw_bytes = audio_obj.read()
                        except Exception:
                            raw_bytes = None

                # If we have simpleaudio and raw bytes, try to parse WAV and play via play_buffer
                if self._simpleaudio and raw_bytes:
                    try:
                        # check for WAV header
                        if raw_bytes[:4] == b'RIFF' and b'WAVE' in raw_bytes[:12]:
                            bio = io.BytesIO(raw_bytes)
                            try:
                                with wave.open(bio, 'rb') as w:
                                    nchannels = w.getnchannels()
                                    sampwidth = w.getsampwidth()
                                    framerate = w.getframerate()
                                    frames = w.readframes(w.getnframes())
                                # simpleaudio expects: audio_data, num_channels, bytes_per_sample, sample_rate
                                play_obj = self._simpleaudio.play_buffer(frames, nchannels, sampwidth, framerate)
                                with self._playback_lock:
                                    self._current_playback = play_obj
                                    self._stop_event.clear()
                                while play_obj.is_playing() and not self._stop_event.is_set():
                                    time.sleep(0.05)
                                if self._stop_event.is_set():
                                    try:
                                        play_obj.stop()
                                    except Exception:
                                        pass
                                with self._playback_lock:
                                    self._current_playback = None
                                played = True
                            except Exception as e:
                                logger.debug("WAV parse/play failed: %s", e)
                                played = False
                        else:
                            # not a WAV; attempt to play as raw assuming standard params
                            try:
                                # guess: stereo 16-bit 44100
                                play_obj = self._simpleaudio.play_buffer(raw_bytes, 2, 2, 44100)
                                with self._playback_lock:
                                    self._current_playback = play_obj
                                    self._stop_event.clear()
                                while play_obj.is_playing() and not self._stop_event.is_set():
                                    time.sleep(0.05)
                                if self._stop_event.is_set():
                                    try:
                                        play_obj.stop()
                                    except Exception:
                                        pass
                                with self._playback_lock:
                                    self._current_playback = None
                                played = True
                            except Exception as e:
                                logger.debug("simpleaudio fallback play_buffer failed: %s", e)
                                played = False
                    except Exception as e:
                        logger.debug("Interrupted simpleaudio playback path: %s", e)
                        played = False

                if not played:
                    # fallback to SDK's play (blocking)
                    try:
                        with self._playback_lock:
                            self._current_playback = "sdk_play"
                            self._stop_event.clear()
                        if self._eleven_play:
                            try:
                                self._eleven_play(audio_obj)
                            except Exception as e:
                                logger.exception("ElevenLabs play failed: %s", e)
                        with self._playback_lock:
                            self._current_playback = None
                    except Exception as e:
                        logger.debug("Fallback SDK play path failed: %s", e)
            except Exception as e:
                logger.exception("Voice worker unexpected error: %s", e)
        logger.debug("Eleven voice worker exited.")

# initialize voice manager if possible
try:

import json as _json  # For spec dumping
import os as _os  # For file operations
from typing import Dict, Any

class AgentUpgrader:
    def __init__(self, agent):
        self.agent = agent
        self.hot_reload_manager = None  # Will be set externally if available

    def set_hot_reload_manager(self, manager):
        """Allow external injection of HotReloadManager"""
        self.hot_reload_manager = manager

    def apply_upgrade(self, upgrade_spec: Dict[str, Any]) -> bool:
        """
        Applies an upgrade to the agent based on the provided spec.
        Now includes hot reloading, XP rewards, and spec logging.
        """
        spec = upgrade_spec.copy()
        upgrade_type = spec.get("type", "unknown")
        self.ui_log(f"Applying upgrade: {upgrade_type}")

        # Show spec in GUI for transparency
        self.ui_log("Upgrade Spec:")
        self.ui_log(_json.dumps(spec, indent=2))

        success = False

        try:
            if upgrade_type == "skill_patch":
                success = self._apply_skill_patch(spec)
            elif upgrade_type == "behavior_mod":
                success = self._apply_behavior_mod(spec)
            elif upgrade_type == "tool_add":
                success = self._add_tool(spec)
            else:
                self.ui_log(f"Unknown upgrade type: {upgrade_type}")
                return False

            if success:
                # Track XP / rewards
                xp_reward = spec.get("xp_reward", 100)
                self.agent.xp += xp_reward
                self.ui_log(f"🏆 +{xp_reward} XP awarded for successful upgrade.")

                # Integrate hot reloading
                if self.hot_reload_manager:
                    code_artifact = spec.get("code_artifact")
                    if code_artifact:
                        self.hot_reload_manager.apply_artifact(code_artifact)
                        self.ui_log("Hot-reloaded new code artifact.")
                    else:
                        self.ui_log("No code artifact to hot-reload.")
                else:
                    # Fallback: write to file if no hot reloader
                    filepath = spec.get("target_file")
                    code = spec.get("code_artifact")
                    if filepath and code:
                        _os.makedirs(_os.path.dirname(filepath), exist_ok=True)
                        with open(filepath, 'w') as f:
                            f.write(code)
                        self.ui_log(f"Upgrade written to {filepath}")

            return success

        except Exception as e:
            self.ui_log(f"Upgrade failed: {str(e)}")
            return False

    def _apply_skill_patch(self, spec: Dict) -> bool:
        skill_name = spec["skill"]
        patch = spec["patch"]
        if hasattr(self.agent, skill_name):
            original = getattr(self.agent, skill_name)
            if callable(original):
                new_func = self._patch_function(original, patch)
                setattr(self.agent, skill_name, new_func)
                self.ui_log(f"Patched skill: {skill_name}")
                return True
        return False

    def _apply_behavior_mod(self, spec: Dict) -> bool:
        key = spec["key"]
        value = spec["value"]
        if hasattr(self.agent, key):
            setattr(self.agent, key, value)
            self.ui_log(f"Modified behavior: {key} = {value}")
            return True
        return False

    def _add_tool(self, spec: Dict) -> bool:
        tool_name = spec["name"]
        tool_code = spec["code"]
        # Simulate dynamic tool registration
        exec(tool_code, globals())
        tool_func = globals().get(tool_name)
        if tool_func:
            self.agent.tools[tool_name] = tool_func
            self.ui_log(f"Added new tool: {tool_name}")
            return True
        return False

    def _patch_function(self, original, patch):
        """Simple string-based function patching (for demo)"""
        # In real use, compile patch and merge
        # This is a stub — real patching would use AST or bytecode
        exec(patch, globals())
        return globals()[original.__name__]

    def ui_log(self, message: str):
        """Centralized logging to UI"""
        if hasattr(self.agent, 'ui_log'):
            self.agent.ui_log(message)
        else:
            print(f"[Upgrade] {message}")

    def _safe_call(self, fn: Callable, *a, **kw):
        try:
            if threading.current_thread() is threading.main_thread():
                return fn(*a, **kw)
            else:
                self.root.after(0, lambda: fn(*a, **kw))
        except Exception:
            logger.exception("Safe call failed")

    def _safe_add_system(self, message: str):
        self._safe_call(self.add_system_message, message)

    def _execute_pending_upgrade(self, user_request: str):
        """
        Full self-upgrade pipeline:
        1. Generate structured spec
        2. Generate code via CodeForge
        3. Validate in sandbox
        4. Log success/failure to learning journal
        5. Write preview file
        6. UPGRADE: Apply dynamic upgrades via AgentUpgrader if possible
        """
        try:
            self.ui_log("Generating structured specification...")
            if not SpecGenerator:
                self.ui_log("SpecGenerator not available.")
                return
            spec_gen = SpecGenerator(self.api_key)
            spec = spec_gen.create(user_request)
            self.ui_log(f"Spec created: {spec.get('feature_name', 'Unnamed feature')}")

            self.ui_log("Invoking CodeForge...")
            if not self.sess.codeforge:
                self.ui_log("CodeForge not initialized.")
                return
            result = self.sess.codeforge.generate_from_spec(spec)

            if not result.get("success", False):
                errors = result.get("errors", [])
                for err in errors:
                    self.ui_log(f"{err}")
                self.ui_log("Logging failure to learning journal...")
                if self.sess.upgrade_journal:
                    self.sess.upgrade_journal.log_failure(spec, result.get("code", ""), {"passed": False, "errors": errors})
                return

            self.ui_log("Running sandbox validation...")
            if not self.sess.validator:
                self.ui_log("SandboxValidator not available.")
                return
            validation = self.sess.validator.test(result["code"], spec)
            if not validation.passed:
                self.ui_log("Post-validation caught issues:")
                for err in validation.errors:
                    self.ui_log(f"  • {err}")
                if self.sess.upgrade_journal:
                    self.sess.upgrade_journal.log_failure(spec, result["code"], validation.to_dict())
                return
            else:
                self.ui_log("Code passed sandbox validation.")

            # UPGRADE: Attempt dynamic application via AgentUpgrader
            try:
                upgrade_spec = {
                    "type": spec.get("upgrade_type", "code_patch"),  # Assume type from spec
                    "code_artifact": result["code"],
                    "target_file": f"upgrades/{spec.get('feature_name', 'feature')}.py",
                    "xp_reward": 150,  # Increased reward for complex upgrades
                    # Add more fields as needed from spec
                }
                if self.agent_upgrader.apply_upgrade(upgrade_spec):
                    self.ui_log("Dynamic upgrade applied successfully via AgentUpgrader.")
                else:
                    self.ui_log("Dynamic upgrade failed; falling back to preview file.")
                    raise Exception("Dynamic apply failed")
            except Exception:
                # Fallback: Write preview file
                try:
                    safe_name = re.sub(r'[^\w\-]', '_', spec.get('feature_name', 'upgrade'))
                    file_name = f"upgrade_preview_{safe_name}.py"
                    with open(file_name, "w", encoding="utf-8") as f:
                        f.write(result["code"])
                    self.ui_log(f"Code written to {file_name}")
                except Exception as write_err:
                    self.ui_log(f"Failed to write preview: {write_err}")

            if self.sess.upgrade_journal:
                self.sess.upgrade_journal.log_success(spec, result["code"])
            self.ui_log("Upgrade logged successfully.")
            self.ui_log("Self-upgrade complete!")

        except Exception as e:
            self.ui_log(f"Error during upgrade: {e}")
            traceback.print_exc()

        self._safe_add_system(msg)

    def send_message(self, event=None):
        user_input = self.input_field.get().strip()
        if not user_input:
            return
        self.input_field.delete(0, tk.END)
        try:
            if self.sess.ai:
                try:
                    self.sess.ai.increment_conversation_count()
                    self.conv_label.config(text=f"Dialogues: {self.sess.ai.total_conversations}")
                except Exception:
                    pass
        except Exception:
            pass
        owner = "You"
        try:
            if self.sess.ke:
                owner = get_preference(self.sess.ke.conn, 'owner_name') or "You"
        except Exception:
            pass
        self.add_message(owner, user_input)
        try:
            if self.sess.voice_enabled and VOICE_AVAILABLE:
                try:
                    vm = get_voice()
                    vm.stop_playback()
                except Exception:
                    pass
        except Exception:
            pass

        # NEW: Detect upgrade commands
        low = user_input.lower().strip()
        if low.startswith("upgrade:") or low.startswith("add feature:") or low.startswith("improve:"):
            threading.Thread(target=self._execute_pending_upgrade, args=(user_input,), daemon=True).start()
            return

        threading.Thread(target=self.process_input, args=(user_input,), daemon=True, name="ProcessInput").start()

    def _handle_pending_spec_confirmation(self, user_input: str, ui_log) -> Optional[str]:
        low = user_input.lower().strip()
        if not self.sess._pending_spec:
            return None
        if low.startswith('apply'):
            parts = low.split()
            token = parts[1] if len(parts) > 1 else None
            safety = self.sess._pending_spec.get('safety_level', 'high')
            if safety != 'low':
                owner_token = os.getenv('SELF_MOD_OWNER_TOKEN')
                if not token or not owner_token or token != owner_token:
                    return "This proposal requires owner authorization. Use: apply <OWNER_TOKEN>"
            ui_log("I will create a proposal now...")
            try:
                if not self.sess.self_mod:
                    ui_log("I cannot modify my code here; self-mod engine is unavailable.")
                    self.sess._pending_spec = None
                    return "Self-modification engine not available."
                res = self.sess.self_mod.accept_spec_and_propose(self.sess._pending_spec, actor='user')
                self.sess._pending_spec = None
                return f"I created a proposal: {res}"
            except Exception as e:
                self.sess._pending_spec = None
                return f"I failed to create a proposal: {e}"
        if low in ['cancel', 'abort', 'nevermind', 'stop']:
            self.sess._pending_spec = None
            return "I cancelled the pending request."
        return None

    def process_input(self, user_input: str):
        try:
            if not self.sess.ai or not self.ultra_processor:
                self._safe_add_system("I'm not ready yet — my core is still starting.")
                return

            def _show_typing():
                try:
                    self.conversation.config(state=tk.NORMAL)
                    self.conversation.insert(tk.END, "Saraphina is thinking...\n", "typing")
                    if self._autoscroll:
                        self.conversation.see(tk.END)
                    self.conversation.config(state=tk.DISABLED)
                except Exception:
                    pass
            self._safe_call(_show_typing)
            time.sleep(0.6 + random.uniform(0.0, 0.6))

            def ui_log(msg):
                self.add_system_message(msg)
            pending_resp = self._handle_pending_spec_confirmation(user_input, ui_log)
            if pending_resp:
                self._remove_last_line_if_typing()
                self.add_message("Saraphina", pending_resp)
                return

            try:
                if parse_and_prepare_proposal:
                    parse_result = parse_and_prepare_proposal(self.sess, user_input, dry_run=True)
                else:
                    parse_result = {}
                spec = parse_result.get("spec") if isinstance(parse_result, dict) else None
                if spec and spec.get("intent"):
                    if spec.get("clarifying_questions"):
                        for q in spec["clarifying_questions"]:
                            self.add_system_message(f"I need more info: {q}")
                        self.sess._pending_spec = spec
                        self._remove_last_line_if_typing()
                        return
                    else:
                        summary = parse_result.get("improvement_spec", "")[:2000]
                        self.add_system_message("I can propose this change:\n" + summary)
                        self.add_system_message("Type 'apply <OWNER_TOKEN>' to create a proposal (owner token required for non-low-risk), or 'cancel' to ignore.")
                        self.sess._pending_spec = spec
                        self._remove_last_line_if_typing()
                        return
            except Exception:
                logger.debug("parse_and_prepare_proposal failure; continuing")

            response = self.ultra_processor.process_query_with_ultra(user_input, ui_callback=ui_log)
            self._remove_last_line_if_typing()
            self.add_message("Saraphina", response)
            if self.sess.voice_enabled:
                speak_text(response)
            self.update_status()
        except Exception as e:
            logger.exception("Processing error: %s", e)
            self._safe_add_system(f"I encountered an error while processing: {e}")

    def _remove_last_line_if_typing(self):
        def _remove():
            try:
                self.conversation.config(state=tk.NORMAL)
                full = self.conversation.get("1.0", tk.END)
                marker = "Saraphina is thinking...\n"
                idx = full.rfind(marker)
                if idx != -1:
                    pre = full[:idx]
                    line_no = pre.count('\n') + 1
                    last_newline = pre.rfind('\n')
                    if last_newline == -1:
                        char_no = idx
                    else:
                        char_no = idx - last_newline - 1
                    start_index = f"{line_no}.{char_no}"
                    try:
                        found_index = self.conversation.search(marker, "1.0", tk.END, backwards=True)
                        if found_index:
                            end_pos = f"{found_index} + {len(marker)}c"
                            self.conversation.delete(found_index, end_pos)
                    except Exception:
                        new_text = full[:idx] + full[idx + len(marker):]
                        self.conversation.delete("1.0", tk.END)
                        self.conversation.insert(tk.END, new_text)
                self.conversation.config(state=tk.DISABLED)
            except Exception as e:
                logger.debug("Failed to remove typing indicator: %s", e)
        self._safe_call(_remove)

    def start_voice_listening(self):
        if not self.sess.stt or not getattr(self.sess.stt, 'available', False):
            return
        try:
            wake_word = get_preference(self.sess.ke.conn, 'wake_word') or 'saraphina'
        except Exception:
            wake_word = 'saraphina'

        def voice_handler(text: str):
            try:
                self.is_listening = True
                self.update_status()
                if VOICE_AVAILABLE:
                    try:
                        get_voice().stop_playback()
                    except Exception:
                        pass
                try:
                    owner = get_preference(self.sess.ke.conn, 'owner_name') or 'You'
                except Exception:
                    owner = 'You'
                self.add_message(f"{owner} (voice)", text)
                if text.lower().startswith(('remind me', 'set a reminder')):
                    m = re.search(r"remind me to (.+) at (\d{1,2}:\d{2})", text, re.IGNORECASE)
                    if m and self.sess.mm:
                        what, when = m.group(1), m.group(2)
                        try:
                            self.sess.mm.add_reminder(what.strip(), datetime.now().strftime('%Y-%m-%d ') + when)
                            response = f"I will remind you to {what} at {when}."
                            self.add_message("Saraphina", response)
                            if self.sess.voice_enabled:
                                speak_text(response)
                        except Exception:
                            self._safe_add_system("I failed to schedule your reminder.")
                        self.is_listening = False
                        self.update_status()
                        return
                if self.sess._pending_spec:
                    low = text.lower().strip()
                    if any(word in low for word in ['apply','yes','proceed','do it','go ahead','confirm','okay','ok']):
                        safety = self.sess._pending_spec.get('safety_level', 'high')
                        if safety == 'low':
                            pending_resp = self._handle_pending_spec_confirmation('apply', lambda m: self._safe_add_system(m))
                            if pending_resp:
                                self.add_message("Saraphina", pending_resp)
                                self.is_listening = False
                                self.update_status()
                                return
                        else:
                            self._safe_add_system("I cannot accept voice approval for high-risk changes. Please authorize with your owner token in chat.")
                            self.is_listening = False
                            self.update_status()
                            return
                    elif any(word in low for word in ['cancel','abort','stop','nevermind']):
                        self.sess._pending_spec = None
                        self.add_message("Saraphina", "I cancelled the pending request.")
                        self.is_listening = False
                        self.update_status()
                        return
                if getattr(self, 'ultra_processor', None):
                    response = self.ultra_processor.process_query_with_ultra(text, ui_callback=lambda m: self._safe_add_system(m))
                    self.add_message("Saraphina", response)
                    if self.sess.voice_enabled:
                        speak_text(response)
                self.is_listening = False
                self.update_status()
            except Exception as e:
                logger.exception("Voice handler error: %s", e)
                self.is_listening = False
                self.update_status()

        try:
            self.sess.stt.start_background(voice_handler, engine='whisper', wake_word=wake_word)
            self._safe_add_system(f"I will listen for '{wake_word}' when I'm allowed.")
        except Exception as e:
            self._safe_add_system(f"I couldn't start background listening: {e}")

                try:
                    vm = get_voice()
                    vm.stop_playback()
                    vm.cleanup()
                except Exception:
                    pass
            if self.sess.ai:
                try:
                    self.sess.ai._save_state()
                except Exception:
                    pass
            self._save_config()
            logger.info("I am shutting down. Goodbye.")
        except Exception as e:
            logger.exception("Shutdown error: %s", e)
        try:
            time.sleep(0.05)
            self.root.destroy()
        except Exception:
            pass

    # ------------------------
    # background learning loops
    # ------------------------
    def start_background_learning(self):
        def learning_loop():
            while self.animation_running:
                try:
                    sleep_time = 300 + random.uniform(-30, 30)
                    time.sleep(max(10, sleep_time))
                    if self.sess.metaopt:
                        health = self.sess.metaopt.analyze_learning_health()
                        if health.get('overall_health') == 'poor':
                            self._safe_add_system("I'm analyzing learning patterns to improve myself.")
                        proposals = self.sess.metaopt.propose_optimizations()
                        if proposals:
                            self._safe_add_system(f"I found {len(proposals)} opportunities to improve.")
                except Exception as e:
                    logger.debug("Learning loop error: %s", e)
        def memory_consolidation():
            while self.animation_running:
                try:
                    sleep_time = 3600 + random.uniform(-60, 60)
                    time.sleep(max(60, sleep_time))
                    if self.sess.mm:
                        try:
                            added = self.sess.mm.consolidate_daily()
                            if added > 0:
                                self._safe_add_system(f"I consolidated {added} memories into knowledge.")
                        except Exception:
                            pass
                except Exception as e:
                    logger.debug("Memory consolidation error: %s", e)
        def health_monitor():
            while self.animation_running:
                try:
                    sleep_time = 1800 + random.uniform(-30, 30)
                    time.sleep(max(30, sleep_time))
                    if self.sess.ke:
                        try:
                            cur = self.sess.ke.conn.cursor()
                            cur.execute("SELECT COUNT(*) FROM facts")
                            fact_count = cur.fetchone()[0]
                            if fact_count < 100:
                                self._safe_add_system("I'm still learning; I need more conversations to grow.")
                            elif fact_count % 100 == 0:
                                self._safe_add_system(f"Milestone reached: {fact_count} facts learned!")
                        except Exception:
                            pass
                except Exception as e:
                    logger.debug("Health monitor error: %s", e)
        threading.Thread(target=learning_loop, daemon=True).start()
        threading.Thread(target=memory_consolidation, daemon=True).start()
        threading.Thread(target=health_monitor, daemon=True).start()
        self._safe_add_system("My background learning is active — I'm continuously improving.")

    # ------------------------
    # small UX helpers
    # ------------------------
    def _register_shortcuts(self):
        try:
            self.root.bind("<Escape>", lambda e: self._on_escape())
            self.root.bind("<Control-s>", lambda e: self._toggle_autoscroll())
        except Exception:
            pass

    def _on_escape(self):
        try:
            if VOICE_AVAILABLE:
                get_voice().stop_playback()
                self._safe_add_system("Playback interrupted.")
        except Exception:
            pass

    def _toggle_autoscroll(self):
        self._autoscroll = not self._autoscroll
        self._safe_add_system(f"Autoscroll {'enabled' if self._autoscroll else 'disabled'}.")
